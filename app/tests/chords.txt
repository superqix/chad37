# M4 Chord mode, Scale mode, SEQ REC pad assignment, Auto-Strum, Arpeggiator, and Global menu test.
#   BOARD=smk37 python3 base/tools/build.py app --host
#   app/build/smk37/host/chad37-smk37_host app/tests/chords.txt OUTDIR
wait 1200
expect preset 1
expect chord_on 0
expect scale_on 0
expect arp_on 0
ble connect
wait 30
out

# ---------------------------------------------------------------------------
# 1. CHORD on, SCALE off: manual fingered chord detection + SEQ REC pad assign
# ---------------------------------------------------------------------------
tap ENV
wait 30
expect chord_on 1
expect scale_on 0
expect pad_chord0 0

# Play C4-E4-G4 (keys 12, 16, 19 -> notes 60=0x3C, 64=0x40, 67=0x43)
keybed 12 down
keybed 16 down
keybed 19 down
wait 30
expect latched_count 3
expect_out USB 90 3C 64
expect_out USB 90 40 64
expect_out USB 90 43 64
shot chord_manual.png
out

# Staggered release preserves the 3-note latched chord
keybed 19 up
wait 10
keybed 16 up
wait 10
keybed 12 up
wait 30
expect latched_count 3
expect active_notes 0
out

# Arm SEQ REC and tap Pad 0 to assign the latched C Major chord
tap REC
wait 30
expect rec_arm 1
tappad 0 100
wait 60
expect rec_arm 0
expect pad_chord0 3
expect active_notes 0
out

# Tapping Pad 0 now plays C4-E4-G4 on key_ch (ch 1); unassigned Pad 1 is silent
tappad 0 100
wait 60
expect_out USB 90 3C 64
expect_out USB 90 40 64
expect_out USB 90 43 64
expect_out TRS 90 3C 64
expect_out TRS 90 40 64
expect_out TRS 90 43 64
expect_out USB 80 3C 00
expect_out USB 80 40 00
expect_out USB 80 43 00
expect active_notes 0
out

tappad 1 100
wait 60
expect active_notes 0
out

# LATCH (NOTE REPEAT, host name LFO): a tapped chord pad keeps sounding after it is let go
tap LFO
wait 30
expect latch_on 1
tappad 0 100
wait 100
expect latched_pad 1
expect active_notes 3
expect_out USB 90 3C 64
expect_out USB 90 40 64
expect_out USB 90 43 64
out
# the same pad again stops it, without a new chord
tappad 0 100
wait 60
expect latched_pad 0
expect active_notes 0
expect_out USB 80 3C 00
expect_out USB 80 40 00
expect_out USB 80 43 00
out
# another pad lets it go (here the empty Pad 1, so silence)
tappad 0 100
wait 60
expect active_notes 3
tappad 1 100
wait 60
expect latched_pad 0
expect active_notes 0
out
# LATCH off lets a held chord go
tappad 0 100
wait 60
expect active_notes 3
tap LFO
wait 30
expect latch_on 0
expect active_notes 0
out
# and with LATCH off a tap is a tap again
tappad 0 100
wait 60
expect active_notes 0
out

# ---------------------------------------------------------------------------
# 2. CHORD on, SCALE on: Easy Chord + Scale Root transposition + morphing
# ---------------------------------------------------------------------------
tap SEL
wait 30
expect chord_on 1
expect scale_on 1
expect page 3
# Transpose scale root from C (0) to D (2) via K1
spin K1 2
wait 30
expect scale_root 2
shot scale_page.png
out

# Press C4 (key 12) + B3 (key 11): Easy Chord C Maj7 transposed +2 -> D Maj7 (62=0x3E, 66=0x42, 69=0x45, 73=0x49)
keybed 12 down
keybed 11 down
wait 30
expect latched_count 4
expect_out USB 90 3E 64
expect_out USB 90 42 64
expect_out USB 90 45 64
expect_out USB 90 49 64
out

# Lift B3 (key 11) while holding C4: morphs smoothly to D Maj triad (only 0x49 turns off)
keybed 11 up
wait 30
expect latched_count 3
expect_out USB 80 49 00
expect active_notes 3
out

keybed 12 up
wait 30
expect_out USB 80 3E 00
expect_out USB 80 42 00
expect_out USB 80 45 00
expect active_notes 0
out

# Assign the latched D Major triad to Pad 1
tap REC
wait 30
expect rec_arm 1
tappad 1 100
wait 60
expect rec_arm 0
expect pad_chord1 3
out

# K3 VOICING to 2X (4): every chord plays with the same chord an octave below. The pad keeps the D triad unvoiced
# (3 notes) and plays D3 F#3 A3 (0x32 0x36 0x39) with D4 F#4 A4
spin K3 4
wait 30
expect voicing 4
tappad 1 100
wait 60
expect pad_chord1 3
expect_out USB 90 32 64
expect_out USB 90 36 64
expect_out USB 90 39 64
expect_out USB 90 3E 64
expect_out USB 90 42 64
expect_out USB 90 45 64
expect active_notes 0
out
# Easy Chord on the keys: D Maj7 doubled is 8 notes; the latched chord (what SEQ REC would store) stays 4
keybed 12 down
keybed 11 down
wait 30
expect latched_count 4
expect active_notes 8
expect_out USB 90 32 64
expect_out USB 90 3D 64
expect_out USB 90 49 64
keybed 12 up
keybed 11 up
wait 30
expect active_notes 0
spin K3 -4
wait 30
expect voicing 0
out

# PATCH (HOME) returns to PG_MAIN while keeping SCALE active
tap HOME
wait 30
expect page 0
expect scale_on 1

# ---------------------------------------------------------------------------

# ---------------------------------------------------------------------------

wait 600

expect_out USB 80 3C 00
expect_out USB 90 40 64
pad 0 up
wait 30
expect active_notes 0
out

midi USB F8 00 00
midi USB F8 00 00
midi USB F8 00 00
midi USB F8 00 00
midi USB F8 00 00
midi USB F8 00 00
midi USB F8 00 00
midi USB F8 00 00
midi USB F8 00 00
midi USB F8 00 00
midi USB F8 00 00
wait 20
expect_out USB 90 3C 64
expect_out USB 90 40 64

wait 30

out
