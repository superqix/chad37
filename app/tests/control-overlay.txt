# On the main view a knob or fader that moves shows big over the body: a dial or a fader moving with it, its name,
# value, CC and channel. It closes 1.5 s after the last move. Other pages and the chord view keep the header message.
wait 1200
listen
wait 30
expect ov_kind 0

# K3 (CC 50 ch 6): the dial; the reader still hears it
spin K3 64
wait 30
expect ov_kind 1
expect ov_idx 2
expect_out USB B5 32 40
said K3
wait 200
expect ov_pos 516
shot knob.png

# The dial eases to a jump rather than snapping
spin K3 63
wait 1
expect ov_pos <1000
wait 300
expect ov_pos 1024
shot knob-full.png

# Closes 1.5 s after the last move
wait 1400
expect ov_kind 0
shot main.png

# A fader: F2 (CC 21); another control replaces the knob at once
fader 1 0
wait 30
spin K1 10
wait 30
expect ov_kind 1
fader 1 96
wait 300
expect ov_kind 2
expect ov_idx 1
expect_out USB B0 15 60
expect ov_pos 774
shot fader.png

# Fader bank 2: F6
tap FADER_BANK
wait 30
fader 1 40
wait 300
expect ov_kind 2
expect ov_idx 5
wait 1600
expect ov_kind 0

# Knob bank 2: K11 (CC 104 ch 1)
tap KNOB_BANK
wait 30
spin K3 5
wait 300
expect ov_kind 1
expect ov_idx 10
expect_out USB B0 68 05
tap KNOB_BANK
wait 1600
expect ov_kind 0

# Another page: no overlay, the header's message
tap GLO
wait 30
spin K3 1
wait 30
expect ov_kind 0
tap HOME
wait 30

# The chord view: no overlay
tap ENV
wait 30
expect chord_on 1
spin K2 3
wait 30
expect ov_kind 0
